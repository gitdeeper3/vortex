#!/usr/bin/env python
"""
نظام Vortex الكامل
"""
import os
import sys
from datetime import datetime

# إضافة المسارات
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from report_manager import VortexReportManager

def main():
    print("🌪️ VORTEX SYSTEM - نظام التنبؤ بالأعاصير")
    print("=" * 50)
    
    # 1. تهيئة النظام
    print("\n1️⃣  تهيئة مدير التقارير...")
    manager = VortexReportManager()
    
    # 2. إنشاء تقارير لأنواع مختلفة
    print("\n2️⃣  إنشاء التقارير...")
    
    basins = ["Atlantic", "Pacific", "Indian"]
    for basin in basins:
        print(f"   📊 إنشاء تقرير لـ {basin}...")
        report_path = manager.generate_forecast_report(basin)
        print(f"   ✅ تم: {os.path.basename(report_path)}")
    
    # 3. عرض التقارير المنشأة
    print("\n3️⃣  التقارير المنشأة:")
    reports_dir = os.path.join(os.path.dirname(__file__), "reports", "daily")
    if os.path.exists(reports_dir):
        reports = os.listdir(reports_dir)
        for report in sorted(reports)[-3:]:  # آخر 3 تقارير
            print(f"   📄 {report}")
    
    print(f"\n✅ تم إنشاء {len(basins)} تقارير بنجاح!")
    print(f"📁 الموقع: {manager.reports_dir}")

if __name__ == "__main__":
    main()
